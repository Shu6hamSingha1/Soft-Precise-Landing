% TikZ block diagram of the DF-ASMC control pipeline.
% Left-to-right cascade: Vision (top) -> Outer Loop (visibility + tracking)
% -> Inner Loop -> Plant. Feedback wraps far-left edge.
% IEEE TAES styling: compact spacing, \small block text (matches body),
% \footnotesize edge labels, \scriptsize eq-refs; unified gray/blue palette
% with gold novelty accent. Fragment: \input{...} inside a figure*.
% Requires: \usepackage{tikz}, \usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds}
% Novel blocks (absent from \cite{singhal2025}): gold border + pale-yellow fill.
%
\begin{tikzpicture}[
    >=Latex,
    font=\small,
    node distance=2mm and 3mm,
    block/.style={
        rectangle, rounded corners=1.2pt, draw=black, thick,
        minimum height=5.2mm, minimum width=17mm,
        align=center, fill=white, inner sep=1.8pt,
        font=\small
    },
    sens/.style ={block, fill=gray!10,   draw=gray!60!black},
    ctrl/.style ={block, fill=blue!8,    draw=blue!60!black},
    plant/.style={block, fill=gray!22,   draw=gray!60!black},
    novel/.style={line width=0.45mm, draw=orange!75!black, fill=yellow!22},
    sum/.style={
        circle, draw=black, thick, inner sep=0pt, minimum size=3.6mm,
        font=\scriptsize
    },
    flow/.style={->, thick, line width=0.3mm},
    ref/.style ={->, thick, line width=0.3mm, dashed},
    dist/.style={->, thick, line width=0.22mm, dashed, dash pattern=on 1pt off 1pt},
    edgelbl/.style={font=\footnotesize, inner sep=1pt},
    eqlbl/.style={font=\scriptsize\itshape, inner sep=0.5pt, text=black!70},
    signlbl/.style={font=\footnotesize, inner sep=0pt},
    grouplbl/.style={font=\footnotesize\bfseries, inner sep=1pt}
]

% =============================================================
% ROW 1 — Vision front-end
% =============================================================
\node[sens]                  (cam)  {Camera};
\node[sens, right=of cam]    (img)  {Image};
\node[sens, right=of img]    (feat) {Image Features};
\node[sens, right=of feat]   (vh)   {Optic Flow};

% =============================================================
% ROW 2 — Outer loop: visibility chain
% =============================================================
\node[sum,   below=9mm of img]                (e1)  {};
\node[ctrl, novel, right=of e1,
      label={[eqlbl]below:Eq.~\eqref{symmetric transformation: equation}}] (fnl1) {Visibility Funnel};
\node[ctrl, novel, right=of fnl1,
      label={[eqlbl]below:Eq.~\eqref{zeta_e_dot_d: equation}}]             (pid)  {Barrier PID};
\node[left=5mm of e1]                         (sd)  {$\hat{\boldsymbol{r}}_\text{d}$};

% =============================================================
% ROW 3 — Outer loop: optic-flow tracking chain
% =============================================================
\node[sum,   below=10mm of pid]               (e2)  {};
\node[ctrl, novel, right=of e2,
      label={[eqlbl]below:Eq.~\eqref{optical flow error transformation: equation}}] (fnl2) {Optic-Flow Funnel};
\node[ctrl, novel, right=of fnl2,
      label={[eqlbl]below:Eq.~\eqref{sigma derivative: equation 2}}]                (sld)  {Sliding Surface};
\node[ctrl, novel, right=of sld,
      label={[eqlbl]below:Eqs.~\eqref{adaptive control law: equation},\,\eqref{adaptive law: equation}}] (asmc) {Adaptive SMC};
\node[ctrl, novel, right=of asmc,
      label={[eqlbl]below:Eq.~\eqref{cone clamp: equation}}]                         (cone) {Cone Clamp};
\node[ctrl, right=of cone,
      label={[eqlbl]below:$\tau_\text{a}=0.08$~s}]                                   (lpf)  {LP Filter};

% =============================================================
% ROW 4 — Inner loop
% =============================================================
\node[ctrl, novel, below=12mm of asmc,
      label={[eqlbl]below:Eq.~\eqref{psi d integrator: equation}}]                   (yaw) {Yaw ASMC};
\node[ctrl, novel, below=12mm of lpf,
      label={[eqlbl]below:Eq.~\eqref{so3 torque: equation}}]                         (so3) {Geometric SO(3)};

% =============================================================
% ROW 5 — Plant
% =============================================================
\node[plant, below=11mm of yaw]               (mix) {Motor Mixing};
\node[plant, below=11mm of so3]               (uav) {UAV Dynamics};

% =============================================================
% CONNECTIONS
% =============================================================

% Vision chain
\draw[flow] (cam)  -- (img);
\draw[flow] (img)  -- (feat);
\draw[flow] (feat) -- (vh);

% --- Visibility side ---
\draw[ref]  (sd) -- (e1);
\draw[flow] ([xshift=-3mm]feat.south) |- (e1.north);
\node[signlbl] at ([xshift=-2.5mm, yshift=-1.5mm]e1) {$\scriptstyle+$};
\node[signlbl] at ([xshift=1.7mm,  yshift=2.3mm]e1)  {$\scriptstyle-$};

\draw[flow] (e1)   -- (fnl1) node[edgelbl, midway, above, yshift=0.3mm]{$\hat{\boldsymbol{r}}_\text{e}$};
\draw[flow] (fnl1) -- (pid)  node[edgelbl, midway, above, yshift=0.3mm]{$\boldsymbol{\zeta}_1$};

% --- Optic-flow side ---
% h_d from PID drops to e2 (dashed — desired optic flow); label on LEFT of arrow
\draw[ref] (pid.south) -- (e2.north)
    node[edgelbl, pos=0.45, left, xshift=-0.5mm]{$\boldsymbol{h}_\text{d}$};

% h from Optic Flow: right along top, down past pid, left into e2.east
\draw[flow] (vh.south) -- ++(0,-3mm) -| ([xshift=2.5mm]pid.east) |- (e2.east);
\node[edgelbl] at ([xshift=4.5mm, yshift=-3mm]vh.south) {$\boldsymbol{h}$};

\node[signlbl] at ([xshift=-1.7mm, yshift=2.5mm]e2) {$\scriptstyle-$};
\node[signlbl] at ([xshift=2.5mm,  yshift=-1.5mm]e2){$\scriptstyle+$};

\draw[flow] (e2)   -- (fnl2) node[edgelbl, midway, above, yshift=0.3mm]{$\boldsymbol{h}_\text{e}$};
\draw[flow] (fnl2) -- (sld)  node[edgelbl, midway, above, yshift=0.3mm]{$\boldsymbol{\zeta}_2$};

% --- ASMC chain ---
\draw[flow] (sld)  -- (asmc) node[edgelbl, midway, above, yshift=0.3mm]{$\boldsymbol{\sigma}$};
\draw[flow] (asmc) -- (cone) node[edgelbl, midway, above, yshift=0.3mm]{$\,^\mathcal{I}\boldsymbol{a}_\text{d}$};
\draw[flow] (cone) -- (lpf);

% --- LP filter -> SO(3): dashed (desired filtered accel) ---
\draw[ref] (lpf.south) -- (so3.north)
    node[edgelbl, midway, right, xshift=0.8mm]{$\,^\mathcal{I}\tilde{\boldsymbol{a}}_\text{d}$};

% --- Yaw ASMC -> SO(3): horizontal dashed ---
\draw[ref] (yaw.east) -- (so3.west)
    node[edgelbl, midway, above, yshift=0.3mm]{$\psi_\text{d}$};

% --- alpha feed: image features -> down -> far-left -> up to yaw ---
\draw[flow] ([xshift=3mm]feat.south) -- ++(0,-2mm)
    -| ([xshift=-4mm]cam.west |- mix.west)
    -| (yaw.south);
\node[edgelbl] at ([xshift=-4mm, yshift=3mm]cam.west |- mix.north) {$\alpha$};

% --- Inner -> plant ---
\draw[flow] (yaw.south) -- (mix.north)
    node[edgelbl, midway, right, xshift=0.5mm]{$\,^\mathcal{B}\boldsymbol{\tau}_u$};
\draw[flow] (so3.south) -- (uav.north);
\draw[flow] (mix) -- (uav);

% --- B_T_u bypass: LP filter -> right -> down -> left to mix.east ---
\draw[flow] (lpf.east) -- ++(4mm,0) |- (mix.east);
\node[edgelbl] at ([xshift=6mm, yshift=-4mm]lpf.east) {$\,^\mathcal{B}T_u$};

% --- Disturbance inputs into plant (single line to avoid vertical bloat) ---
\node[edgelbl, right=6mm of uav, align=center]  (dist)
    {$\boldsymbol{d}_w,\delta m,\delta J,g_\text{e}$};
\draw[dist] (dist.west) -- (uav.east);

% --- State feedback: uav -> down -> far-left -> up to camera ---
\draw[flow] (uav.south) -- ++(0,-4mm)
    -| ([xshift=-9mm]cam.west) -- (cam.west);
\node[edgelbl] at ([xshift=-9mm, yshift=3mm]cam.west |- uav.south) {$\boldsymbol{x}$};

% =============================================================
% Group backdrops — labels placed to avoid overlap with adjacent rows
% =============================================================
\begin{pgfonlayer}{background}
  % VISION FRONT-END: label above, natural center
  \node[draw=gray!60!black, dashed, rounded corners,
        fit=(cam)(img)(feat)(vh),
        inner sep=1mm,
        label={[grouplbl, gray!60!black]above:VISION FRONT-END}] {};
  % OUTER LOOP: label on LEFT shoulder to keep clear of vision row above
  \node[draw=blue!60!black, dashed, rounded corners, line width=0.3mm,
        fit=(sd)(e1)(fnl1)(pid)(e2)(fnl2)(sld)(asmc)(cone)(lpf),
        inner sep=2.8mm,
        label={[grouplbl, blue!60!black, xshift=-48mm, yshift=-1mm]above:OUTER-LOOP CONTROLLER}] {};
  % INNER LOOP: label to RIGHT of the fit box (open area — no overlap)
  \node[draw=blue!60!black, dashed, rounded corners, line width=0.3mm,
        fit=(yaw)(so3),
        inner sep=2.2mm,
        label={[grouplbl, blue!60!black, xshift=0.5mm]right:INNER LOOP}] {};
  % QUADROTOR PLANT: label below
  \node[draw=gray!60!black, dashed, rounded corners,
        fit=(mix)(uav),
        inner sep=1mm,
        label={[grouplbl, gray!60!black]below:QUADROTOR PLANT}] {};
\end{pgfonlayer}

% =============================================================
% Novelty legend — bottom-right, below disturbance label (clear space)
% =============================================================
\node[novel, minimum width=3.5mm, minimum height=2.2mm, inner sep=0]
    at ([xshift=0mm, yshift=-5mm]dist.south) (legbox) {};
\node[anchor=west, font=\scriptsize, right=0.8mm of legbox]
    {Novel vs.\ \cite{singhal2025}};

\end{tikzpicture}
