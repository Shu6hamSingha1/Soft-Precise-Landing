% TikZ block diagram of the MDF-ASMC control pipeline.

% IEEE TAES styling: compact spacing, \small block text, \footnotesize edge

%
\begin{tikzpicture}[
    >=Latex,
    font=\small,
    node distance=2mm and 3mm,
    block/.style={
        rectangle, rounded corners=1.2pt, draw=black, thick,

    sens/.style ={block, fill=gray!10,   draw=gray!60!black},

    novel/.style={line width=0.45mm, draw=orange!75!black, fill=yellow!22},
    sum/.style={
        circle, draw=black, thick, inner sep=0pt, minimum size=3.6mm,
        font=\scriptsize
    },
    flow/.style={->, thick, line width=0.3mm},
    ref/.style ={->, thick, line width=0.3mm, dashed},
    dist/.style={->, thick, line width=0.22mm, dashed, dash pattern=on 1pt off 1pt},
    edgelbl/.style={font=\footnotesize, inner sep=1pt},
    eqlbl/.style={font=\scriptsize\itshape, inner sep=0.5pt, text=black!70},
    signlbl/.style={font=\footnotesize, inner sep=0pt},
    grouplbl/.style={font=\footnotesize\bfseries, inner sep=1pt}
]

% =============================================================
% ROW 1 — Vision front-end
% =============================================================
\node[sens]                                  (cam)  {Camera};
\node[sens, right=6mm of cam]                (img)  {Image};
\node[sens, above right=2mm and 8mm of img]  (feat) {Image\\Features};
\node[sens, below right=2mm and 8mm of img]  (vh)   {Optic Flow};

% =============================================================
% ROW 2 — Image-feature signal chain (PID side)
% =============================================================
\node[sum,  below=10mm of feat]                              (e1)  {};
\node[ctrl, novel, right=10mm of e1, minimum width=22mm,
      label={[eqlbl]below:Eqs.~\eqref{s_e_dot_d: equation},\,\eqref{h_d final: equation}}]
                                                             (pid) {Normalized Virtual\\Feature PID};
\node[left=4mm of e1]                                        (rd)  {$\hat{\boldsymbol{r}}_\text{d}$};

% =============================================================
% ROW 3 — Optic-flow signal chain
% =============================================================
\node[sum,  below=10mm of vh]                                (e2)  {};
\node[ctrl, novel, right=10mm of e2,
      label={[eqlbl]below:Eq.~\eqref{optical flow error transformation: equation}}]
                                                             (ppc)   {PPC};
\node[ctrl, novel, right=of ppc,
      label={[eqlbl]below:Eq.~\eqref{unconstrained system: equation}}]
                                                             (errtr) {Error\\Transformation};

% =============================================================
% ROW 4 — Optic-flow ASMC quartet
% =============================================================
\node[ctrl, novel, below=18mm of e2, minimum width=24mm]                              (sld)
    {Sliding Surface\\[-1pt]{\scriptsize$\boldsymbol{\sigma}=\boldsymbol{\zeta}_2+\mathcal{X}\!\int\!\boldsymbol{\zeta}_2\,d\tau$}};
\node[ctrl, novel, right=of sld, minimum width=30mm]                                  (regr)
    {Regressor\\[-1pt]{\scriptsize$\boldsymbol{\theta}=[-\boldsymbol{c}_h+\mathcal{S}_2\dot{\boldsymbol{p}}_2-\mathcal{G}_2^{-1}\!\mathcal{X}\boldsymbol{\zeta}_2,\,I]$}};
\node[ctrl, novel, right=of regr, minimum width=18mm,
      label={[eqlbl]below:Eq.~\eqref{adaptive law: equation}}]                        (adgn)
    {Adaptive\\Gain};
\node[ctrl, novel, right=of adgn, minimum width=18mm,
      label={[eqlbl]below:Eq.~\eqref{adaptive control law: equation}}]                (claw)
    {Control\\Law};

% =============================================================
% ROW 5 — Cone clamp + LP filter (right of CtrlLaw)
% =============================================================
\node[ctrl, novel, right=4mm of claw,
      label={[eqlbl]below:Eq.~\eqref{cone clamp: equation}}]                          (cone)
    {Funnel-Margin\\Cone Clamp};
\node[ctrl, right=of cone,
      label={[eqlbl]below:Eq.~\eqref{lpf: equation}}]                                 (lpf)
    {LP Filter};

% =============================================================
% ROW 6 — Yaw ASMC sub-loop
% =============================================================
\node[sum,  below=22mm of sld]                                                        (e3)  {};
\node[left=4mm of e3]                                                                 (alpd){$\alpha_\text{d}$};
\node[ctrl, novel, right=10mm of e3, minimum width=24mm]                              (ysld)
    {Yaw Sliding Surface\\[-1pt]{\scriptsize$\sigma_\alpha=\alpha_e+\chi_\alpha\!\int\!\alpha_e\,d\tau$}};
\node[ctrl, novel, below=4mm of ysld, minimum width=20mm,
      label={[eqlbl]below:Eq.~\eqref{yaw control law: equation}}]                     (yadgn)
    {Yaw Adaptive Gain};
\node[ctrl, novel, right=8mm of ysld, minimum width=18mm,
      label={[eqlbl]below:Eq.~\eqref{yaw control law: equation}}]                     (yclaw)
    {Yaw Control\\Law};
\node[ctrl, novel, right=of yclaw, minimum width=22mm,
      label={[eqlbl]below:Eq.~\eqref{psi d integrator: equation}}]                    (vci)
    {Virtual-Compass\\Integrator};

% =============================================================
% ROW 7 — Inner loop (SO(3))
% =============================================================
\node[ctrl, novel, right=of vci, minimum width=22mm,
      label={[eqlbl]below:Eqs.~\eqref{R_d construction: equation}--\eqref{so3 torque: equation}}]
                                                                                      (so3)
    {Geometric SO(3)};

% =============================================================
% ROW 8 — Plant
% =============================================================
\node[plant, below=14mm of yadgn, minimum width=22mm]                                 (mix) {Motor Mixing};
\node[plant, right=of mix, minimum width=22mm]                                        (uav) {UAV Dynamics};

% =============================================================
% CONNECTIONS
% =============================================================

% --- Vision chain ---
\draw[flow] (cam) -- (img);
\draw[flow] (img.east) -- ++(2mm,0) |- (feat.west);
\draw[flow] (img.east) -- ++(2mm,0) |- (vh.west);

% --- Image Features outputs ---
% ^V hat{r} -> e1 (down)
\draw[flow] (feat.south) -- (e1.north)
    node[edgelbl, midway, right, xshift=0.5mm]{$\,^\mathcal{V}\hat{\boldsymbol{r}}$};
% s -> pid (right of feat, down, into pid.north)
\draw[flow] ([xshift=4mm]feat.south) |- ([yshift=3mm]pid.north) -- (pid.north)
    node[edgelbl, pos=0.95, right, xshift=0.5mm]{$\boldsymbol{s}$};

% --- Image-feat sum and PID input ---
\draw[ref]  (rd) -- (e1);
\node[signlbl] at ([xshift=-2.5mm, yshift=-1.5mm]e1) {$\scriptstyle+$};
\node[signlbl] at ([xshift=1.7mm,  yshift=2.3mm]e1)  {$\scriptstyle-$};
\draw[flow] (e1) -- (pid) node[edgelbl, midway, above, yshift=0.3mm]{$\hat{\boldsymbol{r}}_\text{e}$};

% --- PID -> e2 (h_d, dashed); Optic Flow -> e2 (h) ---
\draw[ref] (pid.south) -- ++(0,-3mm) -| (e2.north)
    node[edgelbl, pos=0.05, right, xshift=0.5mm]{$\boldsymbol{h}_\text{d}$};
\draw[flow] (vh.south) -- (e2.north)
    node[edgelbl, midway, right, xshift=0.5mm]{$\boldsymbol{h}$};
\node[signlbl] at ([xshift=-1.7mm, yshift=2.5mm]e2) {$\scriptstyle-$};
\node[signlbl] at ([xshift=2.5mm,  yshift=-1.5mm]e2){$\scriptstyle+$};

% --- Optic-flow funnel chain ---
\draw[flow] (e2)  -- (ppc)   node[edgelbl, midway, above, yshift=0.3mm]{$\boldsymbol{h}_e$};
\draw[flow] (ppc) -- (errtr) node[edgelbl, midway, above, yshift=0.3mm]{$\mathcal{S}_2(\boldsymbol{\zeta}_2)$};

% --- PPC outputs (S_2, p_2_dot) -> Regressor and CtrlLaw ---
\coordinate (ppcbus) at ([yshift=-9mm]ppc.south);
\draw[flow] (ppc.south) -- (ppcbus)
    node[edgelbl, pos=0.55, right, xshift=0.3mm]{$\mathcal{S}_2,\dot{\boldsymbol{p}}_2$};
\draw[flow] (ppcbus) -| ([xshift=-3mm]regr.north);
\draw[flow] (ppcbus) -| ([xshift=-4mm]claw.north);

% --- ErrTrans outputs (zeta_2, G_2) -> Sld, Regressor, CtrlLaw ---
\coordinate (errbus) at ([yshift=-12mm]errtr.south);
\draw[flow] (errtr.south) -- (errbus)
    node[edgelbl, pos=0.6, right, xshift=0.3mm]{$\boldsymbol{\zeta}_2,\mathcal{G}_2$};
\draw[flow] (errbus) -| (sld.north);
\draw[flow] (errbus) -| ([xshift=3mm]regr.north);
\draw[flow] (errbus) -| ([xshift=4mm]claw.north);

% sigma from Sld: along south track, drops UP into AdGn.south and Claw.south
\coordinate (sigbus) at ([yshift=-7mm]sld.south);
\draw[flow] (sld.south) -- (sigbus)
    node[edgelbl, pos=0.5, right, xshift=0.3mm]{$\boldsymbol{\sigma}$};
\draw[flow] (sigbus) -| ([xshift=-3mm]adgn.south);
\draw[flow] (sigbus) -| ([xshift=-3mm]claw.south);

% theta from Regr: short south detour

\draw[flow] (regr.south) -- (thetabus)
    node[edgelbl, pos=0.5, right, xshift=0.3mm]{$\boldsymbol{\theta}$};
\draw[flow] (thetabus) -| ([xshift=3mm]adgn.south);
\draw[flow] (thetabus) -| ([xshift=3mm]claw.south);

% --- CtrlLaw -> Cone, LPF ---
\draw[flow] (claw) -- (cone) node[edgelbl, midway, above, yshift=0.3mm]{$\,^\mathcal{I}\boldsymbol{a}_{\text{d},xy}$};
\draw[flow] (claw.south east) -- ++(2mm,-2mm) -| (lpf.south)
    node[edgelbl, pos=0.05, right, xshift=0.3mm]{$\,^\mathcal{I}\boldsymbol{a}_{\text{d},z}$};
\draw[flow] (cone) -- (lpf) node[edgelbl, midway, above, yshift=0.3mm]{$\,^\mathcal{I}\boldsymbol{a}_{\text{d},xy}$};

% --- Yaw sub-loop: alpha feed from Image Features (long routing around ASMC) ---

    node[edgelbl, pos=0.02, above, xshift=-3mm, yshift=0mm]{$\alpha$};

% --- alpha_d reference and yaw sum signs ---
\draw[ref] (alpd) -- (e3);
\node[signlbl] at ([xshift=-2.5mm, yshift=-1.5mm]e3) {$\scriptstyle+$};
\node[signlbl] at ([xshift=1.7mm,  yshift=2.3mm]e3)  {$\scriptstyle-$};

% --- e3 -> ysld (alpha_e) and e3 -> yclaw (alpha_e) along upper bypass ---
\draw[flow] (e3) -- (ysld) node[edgelbl, midway, above, yshift=0.3mm]{$\alpha_e$};
\draw[flow] ([yshift=4mm]e3.east) -- ++(0,4mm) -| ([yshift=4mm]yclaw.north) -- ([yshift=0mm]yclaw.north)
    node[edgelbl, pos=0.05, above, yshift=0mm]{$\alpha_e$};

% --- ysld outputs: sigma_alpha -> yadgn (down), -> yclaw (right) ---
\draw[flow] (ysld) -- (yadgn) node[edgelbl, midway, right, xshift=0.5mm]{$\sigma_\alpha$};
\draw[flow] (ysld) -- (yclaw) node[edgelbl, midway, above, yshift=0.3mm]{$\sigma_\alpha$};

% --- yadgn -> yclaw (kappa_alpha) ---
\draw[flow] (yadgn.east) -- ++(2mm,0) |- (yclaw.south)
    node[edgelbl, pos=0.05, right, xshift=0.3mm]{$\kappa_\alpha(t)$};

% --- yclaw -> vci ---
\draw[flow] (yclaw) -- (vci) node[edgelbl, midway, above, yshift=0.3mm]{$\dot{\psi}_\text{d}$};

% --- Inner loop signals: vci -> so3 (psi_d), lpf -> so3 (a_d_tilde) ---
\draw[ref] (vci.east) -- (so3.west)
    node[edgelbl, midway, above, yshift=0.3mm]{$\psi_\text{d}$};
\draw[ref] (lpf.south) -- (so3.north)
    node[edgelbl, midway, right, xshift=0.5mm]{$\,^\mathcal{I}\tilde{\boldsymbol{a}}_\text{d}$};

% --- Inner -> plant ---
\draw[flow] (so3.south) -| (uav.north)
    node[edgelbl, pos=0.05, right, xshift=0.5mm]{$\,^\mathcal{B}\boldsymbol{\tau}_u$};

% --- B_T_u bypass: lpf -> right edge -> down -> into mix ---
\draw[flow] (lpf.east) -- ++(4mm,0) |- (mix.east)
    node[edgelbl, pos=0.07, right, xshift=0.3mm]{$\,^\mathcal{B}T_u$};

% --- Disturbance inputs into plant ---
\node[edgelbl, right=4mm of uav, align=center]  (dist)
